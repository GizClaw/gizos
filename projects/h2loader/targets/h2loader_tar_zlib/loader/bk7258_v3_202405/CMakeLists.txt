cmake_minimum_required(VERSION 3.5)

if(DEFINED ENV{H2_REPO_ROOT} AND NOT "$ENV{H2_REPO_ROOT}" STREQUAL "")
  set(H2_REPO_ROOT "$ENV{H2_REPO_ROOT}")
else()
  get_filename_component(H2_REPO_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../../.." ABSOLUTE)
endif()

set(H2_BK_TARGET "bk7258")
set(H2_BK_PARTITION_CSV "${CMAKE_CURRENT_LIST_DIR}/partitions/bk7258/auto_partitions.csv")
file(STRINGS "${H2_BK_PARTITION_CSV}" H2_BK_PRIMARY_CP_ROW REGEX "^primary_cp_app")
file(STRINGS "${H2_BK_PARTITION_CSV}" H2_BK_PRIMARY_AP_ROW REGEX "^primary_ap_app")
file(STRINGS "${H2_BK_PARTITION_CSV}" H2_BK_S_APP_ROW REGEX "^s_app")
foreach(H2_BK_ROW_VAR H2_BK_PRIMARY_CP_ROW H2_BK_PRIMARY_AP_ROW H2_BK_S_APP_ROW)
  string(REGEX MATCH ",,([0-9]+)k," H2_BK_SIZE_MATCH "${${H2_BK_ROW_VAR}}")
  if(NOT H2_BK_SIZE_MATCH)
    message(FATAL_ERROR "invalid BK A/B partition row: ${${H2_BK_ROW_VAR}}")
  endif()
  set("${H2_BK_ROW_VAR}_KIB" "${CMAKE_MATCH_1}")
endforeach()
math(EXPR H2_BK_PRIMARY_WINDOW_KIB "${H2_BK_PRIMARY_CP_ROW_KIB} + ${H2_BK_PRIMARY_AP_ROW_KIB}")
if(NOT H2_BK_PRIMARY_WINDOW_KIB EQUAL H2_BK_S_APP_ROW_KIB)
  message(FATAL_ERROR "BK canonical A and trial B windows must have equal physical size")
endif()
if(NOT H2_BK_PRIMARY_CP_ROW MATCHES ",TRUE,TRUE$" OR
   NOT H2_BK_PRIMARY_AP_ROW MATCHES ",TRUE,TRUE$")
  message(FATAL_ERROR "BK canonical AP/CP rows must be writable for Loader copy-back")
endif()

include($ENV{ARMINO_TOOLS_PATH}/build_tools/cmake/project.cmake)
include("$ENV{H2_BAZEL_COMPONENT_MANIFEST}")

if("${ARMINO_SOC}" STREQUAL "${H2_BK_TARGET}_ap")
  foreach(H2_COMPONENT_DIR IN LISTS H2_BAZEL_AP_COMPONENT_DIRS)
    armino_build_component("${H2_COMPONENT_DIR}")
  endforeach()
elseif("${ARMINO_SOC}" STREQUAL "${H2_BK_TARGET}")
  foreach(H2_COMPONENT_DIR IN LISTS H2_BAZEL_CP_COMPONENT_DIRS)
    armino_build_component("${H2_COMPONENT_DIR}")
  endforeach()
else()
  message(FATAL_ERROR "Unsupported BK7258 execution unit: ${ARMINO_SOC}")
endif()

project(h2_bk_h2loader)
